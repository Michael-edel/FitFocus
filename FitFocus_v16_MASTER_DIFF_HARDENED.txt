/**
 * FITFOCUS v16 — SaaS Hardening Layer
 * Rate limit + KV dedup + usage logging for /api/ai
 *
 * Требования:
 * - Cloudflare Pages → Settings → Bindings → KV Namespace:
 *     Binding name: FITFOCUS_KV
 * - Environment Variables (Secret): GEMINI_API_KEY
 * - (опционально) Secret: IP_HASH_SALT (для анонимизации IP)
 */
export async function onRequestPost({ request, env }) {
  const startedAt = Date.now();

  // ---- Helpers ----
  const json = (obj: any, status = 200, extraHeaders: Record<string, string> = {}) =>
    new Response(JSON.stringify(obj), {
      status,
      headers: {
        "Content-Type": "application/json",
        "Cache-Control": "no-store",
        ...extraHeaders,
      },
    });

  const sha256Hex = async (input: string) => {
    const buf = await crypto.subtle.digest("SHA-256", new TextEncoder().encode(input));
    return Array.from(new Uint8Array(buf)).map(b => b.toString(16).padStart(2, "0")).join("");
  };

  // Cloudflare gives real client IP in this header
  const ipRaw = request.headers.get("CF-Connecting-IP") || "unknown";
  const ipHash = await sha256Hex(`${env.IP_HASH_SALT || "ff"}:${ipRaw}`);

  // ---- Payload size limit (4MB) ----
  const contentLength = request.headers.get("content-length");
  if (contentLength && Number(contentLength) > 4 * 1024 * 1024) {
    return json({ error: { message: "Payload too large" } }, 413);
  }

  let bodyText = "";
  let body: any = null;

  try {
    bodyText = await request.text();
    body = bodyText ? JSON.parse(bodyText) : {};
  } catch {
    return json({ error: { message: "Invalid JSON body" } }, 400);
  }

  // Feature name: либо client передаёт body.feature, либо fallback
  const feature = (typeof body?.feature === "string" && body.feature.trim()) ? body.feature.trim() : "ai";

  // ---- RATE LIMIT ----
  // 1) Cooldown: 1 request / 4 seconds per IP+feature
  const cooldownKey = `rl:cd:4s:${ipHash}:${feature}`;
  const seen = await env.FITFOCUS_KV.get(cooldownKey);
  if (seen) {
    await logUsage(env, { ipHash, feature, status: 429, latency: Date.now() - startedAt, bytesIn: bodyText.length });
    return json({ error: { message: "Rate limit exceeded (cooldown)" } }, 429);
  }
  await env.FITFOCUS_KV.put(cooldownKey, "1", { expirationTtl: 4 });

  // 2) Burst cap: max 20 requests / 10 minutes per IP+feature
  // KV не атомарен, но этого достаточно как “budget guard” на MVP.
  const burstKey = `rl:burst:10m:${ipHash}:${feature}`;
  const current = Number(await env.FITFOCUS_KV.get(burstKey) || "0");
  if (current >= 20) {
    await logUsage(env, { ipHash, feature, status: 429, latency: Date.now() - startedAt, bytesIn: bodyText.length });
    return json({ error: { message: "Rate limit exceeded (burst)" } }, 429);
  }
  await env.FITFOCUS_KV.put(burstKey, String(current + 1), { expirationTtl: 60 * 10 });

  // ---- DEDUP / IDEMPOTENCY CACHE (60s) ----
  const bodyHash = await sha256Hex(bodyText || "{}");
  const dedupKey = `dedup:60s:${feature}:${bodyHash}`;

  const cached = await env.FITFOCUS_KV.get(dedupKey, { type: "json" }) as any | null;
  if (cached && typeof cached === "object" && cached.data) {
    await logUsage(env, { ipHash, feature, status: cached.status || 200, latency: Date.now() - startedAt, bytesIn: bodyText.length, cacheHit: true });
    return new Response(JSON.stringify(cached.data), {
      status: cached.status || 200,
      headers: { "Content-Type": "application/json", "Cache-Control": "no-store", "X-FF-Cache": "HIT" },
    });
  }

  // ---- GEMINI PROXY ----
  if (!env.GEMINI_API_KEY) {
    await logUsage(env, { ipHash, feature, status: 500, latency: Date.now() - startedAt, bytesIn: bodyText.length });
    return json({ error: { message: "GEMINI_API_KEY is not configured on the server." } }, 500);
  }

  // Поддержка как старого формата {model, contents, config}, так и прямого payload для Gemini
  const model = body?.model || "gemini-1.5-flash";
  const url = `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${env.GEMINI_API_KEY}`;

  // Если тело уже похоже на официальный формат Gemini (например, содержит contents),
  // отправим его как есть (кроме feature).
  const { feature: _drop, ...payload } = body ?? {};
  const payloadToSend = (payload && typeof payload === "object") ? payload : {};

  const geminiResp = await fetch(url, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(payloadToSend),
  });

  const data = await geminiResp.json();
  const latency = Date.now() - startedAt;

  // ---- SAVE DEDUP CACHE (60s) ----
  await env.FITFOCUS_KV.put(
    dedupKey,
    JSON.stringify({ status: geminiResp.status, data }),
    { expirationTtl: 60 }
  );

  // ---- USAGE LOG (7 days) ----
  await logUsage(env, { ipHash, feature, status: geminiResp.status, latency, bytesIn: bodyText.length });

  return new Response(JSON.stringify(data), {
    status: geminiResp.status,
    headers: {
      "Content-Type": "application/json",
      "Cache-Control": "no-store",
      "X-FF-Cache": "MISS",
    },
  });
}

// Минимальный usage logging в KV: агрегируем по дню+ipHash+feature,
// чтобы не создавать миллионы ключей “на каждый запрос”.
async function logUsage(
  env: any,
  ev: { ipHash: string; feature: string; status: number; latency: number; bytesIn: number; cacheHit?: boolean }
) {
  try {
    const day = new Date().toISOString().slice(0, 10); // YYYY-MM-DD
    const key = `usage:${day}:${ev.ipHash}:${ev.feature}`;

    const prevRaw = await env.FITFOCUS_KV.get(key);
    const prev = prevRaw ? JSON.parse(prevRaw) : { count: 0, errorCount: 0, totalLatency: 0, totalBytesIn: 0, cacheHits: 0, lastStatus: 0, lastTs: 0 };

    prev.count += 1;
    if (ev.status >= 400) prev.errorCount += 1;
    prev.totalLatency += Number(ev.latency) || 0;
    prev.totalBytesIn += Number(ev.bytesIn) || 0;
    if (ev.cacheHit) prev.cacheHits += 1;
    prev.lastStatus = ev.status;
    prev.lastTs = Date.now();

    await env.FITFOCUS_KV.put(key, JSON.stringify(prev), { expirationTtl: 60 * 60 * 24 * 7 });
  } catch {
    // logging must never break main request
  }
}

====================================================
3) ПОВЕДЕНИЕ (КРАТКО)
====================================================

/api/ai:
- Payload limit: 4MB → 413
- Rate limit per IP+feature:
  - cooldown 1 запрос / 4 секунды → 429
  - burst cap 20 запросов / 10 минут → 429
- Dedup cache:
  - SHA256(body) хранится 60 секунд
  - повтор → отдаём cached response (X-FF-Cache: HIT), Gemini не вызываем
- Usage logging (KV, 7 дней):
  - usage:YYYY-MM-DD:ipHash:feature (агрегировано: count, errors, avg latency можно посчитать)
